- OBJ_VRAM_BASE = 0xc0 << 0x13 = 0x6000000
- ROW_STRIDE = 0x20 (lsls #5)
- ROW_STEP = 0x95 + 1 = 0x96 (r2!=0 -> r5=0xf5+0x95=0x8a, else r5=0x95)
- BG_BASE = 0x0600e000 (DAT_080d4e44)
- BG_COLS = 0xa (movs r2,#0xa)
- pack_ui_state = 0x03005850 (DAT_080d4e40)
