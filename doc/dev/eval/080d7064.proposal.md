## plate comment (中文, ASCII 标点)

- BG2CNT_VAL = 0x1e8a

- VRAM_BG2 = 0x0600e000 (BG2 区域)
- VRAM_BG_CHAR_A = 0x0600d000 (charblock A)
- VRAM_BG3 = 0x0600e800 (BG3 区域)
